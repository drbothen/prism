-- Recipe: Credential Access — Mimikatz Detection -- Tags: T1003, T1059 | MITRE: Credential Access | v1.2 SELECT time, actor.process.name, actor.user.name, target.process.name, file.path, source() AS _src FROM federated WHERE actor.process.name ILIKE '%mimikatz%' OR (target.process.name = 'lsass.exe' AND access.type = 'MemoryRead') OR file.path ILIKE '%.kdbx' SINCE 24h ORDER BY time DESC
