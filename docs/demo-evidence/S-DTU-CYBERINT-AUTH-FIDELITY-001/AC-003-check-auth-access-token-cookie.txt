    pub(crate) fn check_auth(state: &CyberintState, headers: &HeaderMap) -> Result<(), ...> {
        if state.auth_mode() == AuthMode::Reject {
            return Err(Box::new(unauthorized()));
        }
        let token = extract_access_token(headers).ok_or_else(|| Box::new(unauthorized()))?;
        if !state.is_valid_access_token(&token) {
            return Err(Box::new(unauthorized()));
        }
        ...
        Ok(())
    }

check_auth delegates to extract_access_token — which only matches "access_token" cookie.
A "cyberint_session" cookie causes extract_access_token to return None => 401.

AC-003 SATISFIED.
